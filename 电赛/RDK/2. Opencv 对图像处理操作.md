```python
import cv2 #opencv导入 ，一般opencv都是BGR格式 [高度，宽度，通道 (3)]
import numpy as np
from hobot_vio import libsrcampy # RDK官方的包

PIPE_ID = 0
# IMX219原生分辨率
WIDTH = 1920
HEIGHT = 1080
# NV12格式高度是原图的1.5倍
YUV_TOTAL_HEIGHT = int(HEIGHT * 1.5)

cam = libsrcampy.Camera() # 创建camera类
# open_cam填写原生支持的1920×1080
# 需要注意，我们用python写opencv，MIPI摄像头只能读取原生分辨率，这里统一设置1920和1080，后续再用opencv缩放尺寸
ret = cam.open_cam(pipe_id=PIPE_ID, video_index=-1, fps=30, width=WIDTH, height=HEIGHT)
if ret != 0:
    print("摄像头打开失败！")
    exit()

# 设置HSV阈值（蓝色为例，可以自行修改）
lower_hsv = np.array([90, 100, 80])
upper_hsv = np.array([125, 255, 255])

print("摄像头开启成功，开始读取画面")
while True:
    # 获取一帧NV12图像，超时没有获取到图像返回None，拿到画面立刻返回
    img_bytes = cam.get_img(2)
    if img_bytes is None:
        continue  # 容错机制，防止后续代码报错，继续读取下一帧图像
    
    # 将字节数据转为np数组，再转换为OpenCV识别的BGR格式
    yuv_np = np.frombuffer(img_bytes, np.uint8).reshape((YUV_TOTAL_HEIGHT, WIDTH))
    frame_bgr = cv2.cvtColor(yuv_np, cv2.COLOR_YUV2BGR_NV12)
    # 软件缩放到640×380
    frame = cv2.resize(frame_bgr, (640, 380))

    # ==========HSV颜色识别核心代码============
    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV) # 将图片从BRG转为HSV图片
    mask = cv2.inRange(hsv, lower_hsv, upper_hsv) # 查找颜色
    #形态学操作去除噪点
    kernel = np.ones((4, 4), np.uint8)
    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    for cnt in contours:
        area = cv2.contourArea(cnt)
        if area < 60:
            continue
        rect = cv2.minAreaRect(cnt)
        box_points = cv2.boxPoints(rect)
        box = np.int0(box_points)
        print("四个顶点坐标：\n", box)
        cv2.drawContours(frame, [box], 0, (0, 255, 0), 2)
        for point in box:
            cv2.circle(frame, (point[0], point[1]), 3, (0, 0, 255), -1)
    # =========================================

    cv2.imshow("mask", mask)
    cv2.imshow("frame", frame)
    key = cv2.waitKey(1) & 0xFF
    if key == ord('q'):
        break

cam.close_cam()
cv2.destroyAllWindows()
```

## 1.  找mask图

```python
    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV) # 将图片从BRG转为HSV图片

```
****
1. `mask = cv2.inRange(hsv, lower_hsv, upper_hsv)`
- **hsv**：经过`cv2.COLOR_BGR2HSV`转换之后的 HSV 图像，矩阵格式是`(高,宽,3)`。
- **lower_hsv**：`np.array([H最小值, S最小值, V最小值])`，像素低于这个值就判定为背景(0)。
- **upper_hsv**：`np.array([H最大值, S最大值, V最大值])`，像素高于这个值判定为背景(0)。
- 在区间内被判定为255
****
2. ```python
    kernel = np.ones((4, 4), np.uint8)
    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
    ```
    这个是图片的处理，做了先腐蚀后膨胀的开运算
****
3. `contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)`
- 轮廓是由白色（255）区域的边界生成，它寻找白色物体，背景是黑色 (0)
- `contours`：==列表==，里面每一个元素对应一个物体的轮廓；
- `contours[0]`：第 1 个物体所有边界点，格式是`N×1×2`；
- 每个点：`[x, y]`像素坐标；
-  `_`：层级信息，现在项目用不到，用下划线丢弃。
- 后面还有两个模式 RETR_TREE、CHAIN_APPROX_NONE，其中cv2.RETR_EXTERNAL是只保留最外部轮廓，而cv2.RETR_TREE 是保留所有内部轮廓，并记录层级 ==@了解圆环怎么识别==
****
## 2. 绘制
```python
    for cnt in contours:
        area = cv2.contourArea(cnt) # 很重要，如果识别大物体，就调大点
        # 轮廓围起来的闭合区域里面所有像素点数量进行统计，圆环内外两个轮廓，都会统计包含位置
        if area < 60: # 如果像素点小于某个值，就认为是噪点 @了解改为相对面积
            continue
        rect = cv2.minAreaRect(cnt)
        box_points = cv2.boxPoints(rect)
        box = np.int0(box_points) # 将浮点数转为整数，因为像素坐标一定为整数
        print("四个顶点坐标：\n", box)
        cv2.drawContours(frame, [box], 0, (0, 255, 0), 2)
        for point in box:
            cv2.circle(frame, (point[0], point[1]), 3, (0, 0, 255), -1)
```
****
1. `rect = cv2.minAreaRect(cnt)`
求取包围轮廓`cnt`的**最小旋转矩形（可以倾斜，不只是水平竖直方框）**
返回值`rect = ( (中心x,中心y), (宽度w,高度h), 旋转角度angle )`
****
2. `box_points = cv2.boxPoints(rect)`
这个代码会通过算法把rect的表格数据转化为旋转矩形的四个顶点（浮点数），具体怎么算的去问ai，让他举例子
****
3. `cv2.drawContours(frame, [box], 0, (0, 255, 0), 2)`
- frame 就是我要画在那个图像上
- \[box]就是转化后的顶点数组，因为这里要求是轮廓==列表==
- 0这里指的是`contourIdx=0`表示画列表的第几个轮廓，这里只有一个，写0，-1是全选,因为最前面的for循环每次只找一个轮廓的原因
- `color= (0, 255, 0)` 指的是画什么颜色 BGR
- `thickness=2`线条粗细，-1表示填充
****
4. `cv2.circle(frame, (point[0], point[1]), 3, (0, 0, 255), -1)`
这个就是在图形的四个点上画实心圆 ,第二个参数是圆心坐标，第三个参数是半径

# 3. 边缘检测
```python
import cv2 #opencv导入 ，一般opencv都是BGR格式 [高度，宽度，通道 (3)]
import numpy as np
from hobot_vio import libsrcampy # RDK官方的包

PIPE_ID = 0
# IMX219原生分辨率
WIDTH = 1920
HEIGHT = 1080
# NV12格式高度是原图的1.5倍
YUV_TOTAL_HEIGHT = int(HEIGHT * 1.5)

cam = libsrcampy.Camera() # 创建camera类
# open_cam填写原生支持的1920×1080
# 需要注意，我们用python写opencv，MIPI摄像头只能读取原生分辨率，这里统一设置1920和1080，后续再用opencv缩放尺寸
ret = cam.open_cam(pipe_id=PIPE_ID, video_index=-1, fps=30, width=WIDTH, height=HEIGHT)
if ret != 0:
    print("摄像头打开失败！")
    exit()

# 设置HSV阈值（蓝色为例，可以自行修改）
lower_hsv = np.array([90, 100, 80])
upper_hsv = np.array([125, 255, 255])

print("摄像头开启成功，开始读取画面")
while True:
    # 获取一帧NV12图像，超时没有获取到图像返回None，拿到画面立刻返回
    img_bytes = cam.get_img(2)
    if img_bytes is None:
        continue  # 容错机制，防止后续代码报错，继续读取下一帧图像
    
    # 将字节数据转为np数组，再转换为OpenCV识别的BGR格式
    yuv_np = np.frombuffer(img_bytes, np.uint8).reshape((YUV_TOTAL_HEIGHT, WIDTH))
    frame_bgr = cv2.cvtColor(yuv_np, cv2.COLOR_YUV2BGR_NV12)
    # 软件缩放到640×480
    frame = cv2.resize(frame_bgr, (640, 480))
    frame = cv2.GaussianBlur(frame, (3,3), 0)
    # ==========边缘检测核心代码============
    frame = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    edge  = cv2.Canny(frame, 100, 200)
    # =========================================

    cv2.imshow("frame", frame)
    cv2.imshow("edge", edge)
    key = cv2.waitKey(1) & 0xFF
    if key == ord('q'):
        break

cam.close_cam()
cv2.destroyAllWindows()
```

这里我们使用的是Canny边缘检测的算法，具体细节去看[[OpenCV基础操作#4. 边缘检测流程（Canny）]] 这里只讲怎么用
用法为`edge = cv2.Canny(gray, threshold1=50, threshold2=150)`

假设：`threshold1=50，threshold2=150（刚好3倍，符合2‑3倍经验规则）`
- 梯度＞150：确定边缘，白线；
- 梯度＜50：黑色；
- 50‑150 之间：
    - 如果旁边连着大于 150 的边缘 → 显示白线；
    - 周围全部都是低于 50 的像素 → 当作噪声，变成黑色。

注意，Canny在opencv的算法里不自带高斯滤波，需要自己写，写在转完gary图后
