# 1.Atcoder ABC474_C
这个属于大数据 模拟类的题目（位置映射，打标记）
```C++
#include <bits/stdc++.h>
#define repu(e, u) for (int e = head[u]; e; e = nxt[e])
#define rep(i,a,b) for (int i = a;i <= b;i++)
using namespace std;
using ll = long long;
const int N = 2e5 + 5;
int n,m;
vector<int> a(N * 3);
unordered_map<int,int> v_idx;
void solve() {
    cin >> n >> m;
    rep(i,1,n) cin >> a[i],v_idx[a[i]] = i;
    rep(i,1,m) {
        int val;cin >> val;
        int idx = v_idx[val];
        a[idx] = 0;a[n + i] = val;
        v_idx[val] = n + i;
    }
    rep(i,1,n + m) {
        if (a[i] != 0) cout << a[i] << " ";
    }
    cout << "\n";
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
```

- `#include <bits/stdc++.h>`这个是万能头文件，包含很多常用头文件，平时刷刷题可以写，正式情况还是自己慢慢写吧

- `#define int long long`可以在做算法题的时候使用，这个时候所有的int都会被替换为long long防止溢出。但是，我们必须写的`int main()`就必须换一种写法了，写成`signed main()`，`signed` 就是`signed int`，不会被宏替换（因为C++要求main必须范围int）

- `#define rep(i,a,b) for(int i=a;i<=b;i++)`这个也可以宏替换，避免了写for循环(宏定义没有分号结尾)

- `using ll = long long`，C++语法，ll来代替long long  

- `rep(i,1,n) cin >> a[i],v_idx[a[i]] = i`这个就可以利用值来找下标，`map`的查找复杂度是$O(\log n)$，题目中用的是`unordered_map`，底层是哈希表，最优是$O(1)$，map的优势可能就是在key自动排序了

- 这个代码提供一个思路，这个是Atcoder ABC487_c的题解，我认为这个思路很好，通过开一个很大的vector，然后把最后一次操作的数字在原位 置0，然后放在后面，往复操作，然后最后遍历把0去掉，unordered_map换成map不确定能不能过，看情况

# 2.Atcoder abc475_c @
```C++
#include<bits/stdc++.h>  
#define int long long  
using namespace std;  
  
signed main(){  
    int N,S,L,ans=0;  
    cin>>N>>S>>L;  
    vector<int> A(N); // A[1]~A[N-1] 相邻城镇道路长度  
  
    for(int i=1;i<=N-1;i++) cin>>A[i];  
    vector<int> pos(N+1,0); // pos[i]：城镇i距离城镇1的总距离  
    for(int i=2;i<=N;i++){  
        pos[i] = pos[i-1] + A[i-1];  
    }  
  
    for(int l=1;l<=S;l++){  
        for(int r=S;r<=N;r++){  
            long long dL = pos[S] - pos[l];  
            long long dR = pos[r] - pos[S];  
            long long total = min(dL, dR) + dL + dR;  
            if(total <= L){  
                ans = max(ans, r-l+1);  
            }  
        }  
    }  
    cout << ans << endl;  
    return 0;  
}
```
首先，这个题是一个tag为枚举，前缀和，二分指针的题目

在一个直线上，看看在有限的L内，起点为S，走过的点数量最大，我们可以归结为1种情况，就是先想左走多少路，再向右走多少路（现向右再向左是等价的），并且路到1的距离通过前缀和去计算， 然后通过比较左和右两侧谁的L最小，选择最小的路程走两遍，再计算以前ans和这次的`r-l+1`的max作为新的ans，即可得出答案

# 3. Atcoder abc472_b
```C++
#include<bits/stdc++.h>  
#define int long long  
using namespace std;  
  
signed main(){  
    int N; cin>>N;  
    vector<int> L(1+N);  
    for(int i=1;i<1+N;i++){  
        cin>>L[i];  
    }  
    vector<int> pos(N+1,0);  
    for(int i=1;i<N+1;i++){  
        pos[i]=pos[i-1]+L[i];  
    }  
    int ans=1e18;  
    for(int i=0;i<N;i++){  
        ans = min(ans,abs(pos[N]-pos[i]*2));  
    }  
    cout<<ans;  
    return 0;  
}
```
这个是我在整理完2后立刻自己写的，启发也是比较大，用了前缀和
有N个小棍子（一个大棍子），N-1个折痕，挑一个折痕折断，剩下两个棍子的距离相差最小是多少
我先把棍子分为1到N个标号，然后把每两个折痕间的距离送入L，然后，我把N个小棍子分为N+1个点，用pos表示，求前缀和，`pos[1]-pos[0]`就是第一根棍子的长度，然后通过`pos[i]=pos[i-1]+L[i]`求出所有点的前缀和，然后通过前缀和就能很快的遍历了，时间复杂度是$O(N)$

# 4. Atcoder  abc156_c
```C++
#include<bits/stdc++.h>
using namespace std;

int solve(int N,int ave, vector<int> arr){
    int ans=0;
    for(int i=1;i<N+1;i++){
       ans+=((arr[i]-ave)* (arr[i]-ave));
    }
    return ans;
}

int main(){
    int N; cin>>N;
    int sum=0;
    vector<int> arr(1+N);
    for(int i=1;i<N+1;i++){
        cin>>arr[i];
        sum+=arr[i];
    }
    int ave=sum/N;
    int ans;
    ans = min(solve(N,ave,arr),solve(N,ave+1,arr));
    cout<<ans;
    return 0;
}
```
这个主要积累下思路，题干是一个坐标，坐标整数点上有N个人，选择在点P开启会议，N个人前往P（P强制为整数）参会，每个人都要花费  $(X_i​−P)^2$ ，求N个人花费的min
$\sum (X_i-P)^2=\sum X_i^2 -2P\sum X_i + N P^2$ 
我们把P当作自变量x，这就是一个二次函数，最小值在$P=\frac{\sum X_i}{N}$ ，也就是平均值ave啊
在代码里，ave就是这个最小的P点，由于C++的整除原则，我们去min(ave，ave+1)这两个点去召开会议就好了。

$\boldsymbol{S(P)=\sum_{i=1}^N |X_i-P|}$
这个的话，P是去中位数是最好的。

# leetcode 二分查找 35
```C++
class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        int left=0,right=nums.size()-1;
        while(left<=right){
            int mid=left+(right-left)/2;
            if(nums[mid]==target){
                return mid;
            }
            else if(nums[mid]<target){
                left=mid+1;
            }
            else{
                right=mid-1;
            }
        }
        return left;
    }
};
```
二分查找在有排序的数组中是很好使的，只有$O(logn)$
