在 C++ 中，**绝大多数运算符都可以被重载**，但有以下五个运算符**不允许被重载**: **条件运算符 (`?:`)**  、**成员访问运算符 (`.`)**、**指向成员运算符 (`.*`)**、**作用域解析运算符 (`::`)**、**`sizeof` 运算符


先看这一段代码
```C++
#include<iostream>
using namespace std;
class Per{
    public:
    int m_a;
    int m_b;
};
int main(){
    Per p1;
    p1.m_a=10;p1.m_b=20;
    
    Per p2;
    p2.m_a=10;p2.m_b=20;
    
    Per p3;
    p3.m_a=p1.m_a+p2.m_a;
    p3.m_b=p1.m_b+p2.m_b;
    
    cout<<p3.m_a<<endl;
    cout<<p3.m_b<<endl;
    return 0;
}
```
理解起来并不是什么难的
```C++
#include<iostream>
using namespace std;
class Per{
    public:
    int m_a;
    int m_b;
    Per paddr(Per &p){ //成员函数，以Per为返回类型
	    Per temp;
	    temp.m_a = this->m_a + p.m_a;
	    temp.m_b = this->m_b + p.m_b;
	    return temp;
	}
};
int main(){
    Per p1;
    p1.m_a=10;p1.m_b=20;
    
    Per p2;
    p2.m_a=10;p2.m_b=20;
    
    Per p3;
    p3=p1.paddr(p2)  // p1和p2去使用这个成员函数,以Per的形式返回，然后赋给p3
	
    cout<<p3.m_a<<endl;
    cout<<p3.m_b<<endl;
    return 0;
```
 然后我们对paddr这个方法名字改名为`operator+`
```C++
#include<iostream>
using namespace std;
class Per{
    public:
    int m_a;
    int m_b;
    Per operator+(Per &p){ //成员函数，以Per为返回类型
	    Per temp;
	    temp.m_a = this->m_a + p.m_a;
	    temp.m_b = this->m_b + p.m_b;
	    return temp;
	}
};
int main(){
    Per p1;
    p1.m_a=10;p1.m_b=20;
    
    Per p2;
    p2.m_a=10;p2.m_b=20;
    
    Per p3;
    p3=p1.operator+(p2)  // p1和p2去使用这个成员函数,以Per的形式返回，然后赋给p3
	
    cout<<p3.m_a<<endl;
    cout<<p3.m_b<<endl;
    return 0;
```

此时修改完后，我们就可以把 `p3=p1.operator+(p2)` 改为 `p3=p1+p2` ，这个就叫做运算符的重载
整个重载过程调用了3个类，在p1里去调用重载方法，并且p2传入，以temp暂时类返回赋值给p3



对于双目运算符的重载，重载的时候是成员函数还是友元函数是有很大的区别的，对于成员函数，我们在重载的时候==必须==写一个参数（左操作数已经被本类自带的this占了），对于友元函数，在重载的时候则==必须==写两个参数

```Cpp
#include<iostream>
#include<cmath>
using namespace std;

class Complex{
    int real;
    int image;
public:
    Complex(int r,int i):real(r),image(i){}

    int operator>(Complex c1){
        if(sqrt(real*real +image*image) >sqrt(c1.real*c1.real +c1.image*c1.image) ){
            return 1;
        }
        else return 0;
    }
};

int main(){
    int a,b,c,d;
    while(1){
        cin>>a>>b>>c>>d;
        if(a==0 && b==0 && c==0 && d==0){
            return 0;
        }
        Complex c1(a,b);
        Complex c2(c,d);
        int temp =c1>c2;
        if(temp ==1 ){
            cout << "true" << endl;
        }
        else{
            cout << "false" << endl;
        }
    }
}
```

主要看类部分，我们重载为了成员函数，那么参数表就只能写一个。首先，重载的返回类型是随意的；c1是通过同类访问的特性去访问私有函数的[[类与对象#**同类互访**]]；想要重载什么运算符就把函数名做成什么运算符就可以了，重载是怎么等价的看下面表格

| 表达式      | 成员函数重载            | 全局函数（友元）重载         |
| -------- | ----------------- | ------------------ |
| `a + b`  | `a.operator+(b)`  | `operator+(a, b)`  |
| `a - b`  | `a.operator-(b)`  | `operator-(a, b)`  |
| `a > b`  | `a.operator>(b)`  | `operator>(a, b)`  |
| `a == b` | `a.operator==(b)` | `operator==(a, b)` |
下面写下友元的双目重载(如果是全局的重载那数据就不能写在私有里了，所以一般我们不考虑私有)
```Cpp
#include<iostream>
#include<cmath>
using namespace std;

class Complex{
    int real;
    int image;
public:
    Complex(int r,int i):real(r),image(i){}
    friend int operator>(Complex c1,Complex c2);

};
int operator>(Complex c1,Complex c2){
    if(sqrt(c1.real*c1.real +c1.image*c1.image) >sqrt(c2.real*c2.real +c2.image*c2.image) ){
        return 1;
    }
    else return 0;
}

int main(){
    int a,b,c,d;
    while(1){
        cin>>a>>b>>c>>d;
        if(a==0 && b==0 && c==0 && d==0){
            return 0;
        }
        Complex c1(a,b);
        Complex c2(c,d);
        int temp =c1>c2;
        if(temp ==1 ){
            cout << "true" << endl;
        }
        else{
            cout << "false" << endl;
        }
    }
}
```

参数 = 操作数 - 是否有this占的操作数

| 运算符类型          | 重载为成员函数          | 重载为友元函数                                      |
| -------------- | ---------------- | -------------------------------------------- |
| 一元（如 `-` 负号）   | 0 个参数            | 1 个参数                                        |
| 二元（如 `+`）      | 1 个参数            | 2 个参数                                        |
| 后置 `++` / `--` | 1 个 `int` 参数（占位） | 2 个参数（其中一个是 `int`）（operator++(class &b,int)） |

注意一下，有一些时候重载是需要返回自身的
```Cpp
#include<iostream>
using namespace std;



class Complex
{
    public:
        Complex(double r=0, double i=0):real(r), imag(i){    }
        Complex operator+(Complex temp){
            Complex res;
            res.real = real + temp.real;
            res.imag = imag + temp.imag;
            return res;
        }
        Complex operator-=(Complex temp){
            real -= temp.real;
            imag -= temp.imag;
            return *this;
        }
        friend Complex operator-(Complex temp1,Complex temp2){
            Complex res;
            res.real = temp1.real-temp2.real;
            res.imag = temp1.imag - temp2.imag;
            return res;
        }
        void Display() const;
    private:
        double real;
        double imag;
};

void Complex::Display() const
{
    cout << "(" << real << "," << imag << ")" << endl;
}
int main()
{
    double r, m;
    cin >> r >> m;
    Complex c1(r, m);
    cin >> r >> m;
    Complex c2(r, m);
    Complex c3 = c1+c2;
    c3.Display();
    c3 = c1-c2;
    c3.Display();
    c3 -= c1;
    c3.Display();
    return 0;
}

```
我们关注下 -= 号的重载，需要`return *this`，即返回自身，因为main里面`c3 -= c1`并没有返回值，如果真和别的写的一样的话，return的res会被抛弃
